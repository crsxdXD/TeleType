cmake_minimum_required(VERSION 3.20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(TeleType VERSION 0.2.1 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "TeleType is a Windows Desktop app. Build on Windows or use the GitHub Actions workflow.")
endif()
add_executable(TeleType WIN32 src/main.cpp src/json.hpp src/hotkeys.hpp)
target_compile_features(TeleType PRIVATE cxx_std_17)
target_compile_definitions(TeleType PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(TeleType PRIVATE winhttp ole32 shell32 user32 gdi32)
if(MSVC)
  target_compile_options(TeleType PRIVATE /W4 /utf-8 /EHsc /O1 /Gy /Gw)
  target_link_options(TeleType PRIVATE /OPT:REF /OPT:ICF)
endif()
